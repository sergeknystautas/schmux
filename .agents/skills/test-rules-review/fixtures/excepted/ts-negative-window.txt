import { test, expect } from '@playwright/test'

test('dismissed toast stays dismissed', async ({ page }) => {
  await dismissToast(page)

  // Absence is the claim: the toast must not reappear within the
  // 2000ms window (docs/testing.md rule 4).
  await page.waitForTimeout(2000)

  await expect(page.getByText('Saved')).toBeHidden()
})
