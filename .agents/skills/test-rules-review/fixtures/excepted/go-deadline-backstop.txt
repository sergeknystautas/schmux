package demo_test

import (
	"testing"
	"time"
)

func TestBroadcastDelivered(t *testing.T) {
	select {
	case evt := <-events:
		if evt.Kind != "session-added" {
			t.Fatalf("kind = %q, want session-added", evt.Kind)
		}
	case <-time.After(5 * time.Second):
		t.Fatal("timed out waiting for session-added")
	}
}
