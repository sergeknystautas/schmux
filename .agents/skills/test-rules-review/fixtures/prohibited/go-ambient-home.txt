package demo_test

import (
	"os"
	"path/filepath"
	"testing"
)

func TestReadsUserConfig(t *testing.T) {
	home, err := os.UserHomeDir()
	if err != nil {
		t.Fatal(err)
	}
	cfg := filepath.Join(home, ".schmux", "config.json")
	data, err := os.ReadFile(cfg)
	if err != nil {
		t.Fatal(err)
	}
	if len(data) == 0 {
		t.Fatal("expected config content")
	}
}
