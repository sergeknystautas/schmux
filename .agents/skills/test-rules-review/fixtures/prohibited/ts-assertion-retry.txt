import { test, expect } from '@playwright/test'
import { captureTerminal, captureTmux } from './helpers-terminal'

test('terminal matches tmux', async ({ page }) => {
  const maxRetries = 50
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    let term = ''
    term = await captureTerminal(page)
    const tmux = await captureTmux()
    if (term === tmux) {
      break
    }
    await new Promise((r) => setTimeout(r, 200))
  }
  expect(await captureTerminal(page)).toBe(await captureTmux())
})
